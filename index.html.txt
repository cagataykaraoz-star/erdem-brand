function fmtPct(v, withSign=true){
  if(v===null||v===undefined) return {txt:"N/A", cls:"neu"};

function animateValue(el, end, suffix=""){
  const dur=900, start=performance.now(), from=0;
  function step(now){
    const p=Math.min(1,(now-start)/dur);
    const eased = 1-Math.pow(1-p,3);
    el.textContent = fmtNum(from+(end-from)*eased)+suffix;
    if(p<1) requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}

function heatColor(val, min, max, negOk){
  // teal scale for positive-only, diverging red-teal if negOk
  if(negOk){
    const m = Math.max(Math.abs(min),Math.abs(max),1);
    const t = val/m; // -1..1
    if(t>=0){
      const l = 92 - t*45; // lighter->darker teal
      return `hsl(174, 55%, ${Math.max(30,l)}%)`;
    }else{
      const l = 92 - Math.abs(t)*40;
      return `hsl(4, 65%, ${Math.max(45,l)}%)`;
    }
  }else{
    const t = max===min ? 0.5 : (val-min)/(max-min);
    const l = 88 - t*55;
    return `hsl(198, 55%, ${Math.max(22,l)}%)`;
  }
}

/* ============================= COUNT-UP KPI ANIMATION ============================= */
function animateAllKPIs(container){
  container.querySelectorAll('.value[data-count]').forEach(el=>{
    const val = parseFloat(el.getAttribute('data-count'));
    const suffix = el.getAttribute('data-suffix')||'';
    const start=performance.now(), dur=900;
    function step(now){
      const p=Math.min(1,(now-start)/dur);
      const eased=1-Math.pow(1-p,3);
      const cur = val*eased;
      el.textContent = (suffix==='%' ? (cur>=0?'':'') + cur.toFixed(1) : fmtNum(cur)) + suffix;
      if(p<1) requestAnimationFrame(step);
    }
    requestAnimationFrame(step);
  });
}
animateAllKPIs(document.getElementById('panel-overview'));
COUNTRIES.forEach(c=> animateAllKPIs(document.getElementById('panel-'+c)));

/* ---- Heatmaps (overview) ---- */
function buildHeatmap(containerId, matrix, rows, cols, negOk, valueFmt){
  const el = document.getElementById(containerId);
  const grid = document.createElement('div');
  grid.className='heatmap-grid';
  grid.style.gridTemplateColumns = `140px repeat(${cols.length}, 1fr)`;
  // header row
  grid.appendChild(Object.assign(document.createElement('div'), {className:'hm-header', textContent:''}));
  cols.forEach(cl=>{
    const h=document.createElement('div'); h.className='hm-header'; h.textContent=cl;
    grid.appendChild(h);
  });
  rows.forEach((r,ri)=>{
    const rowVals = matrix[ri];
    const min = Math.min(...rowVals), max = Math.max(...rowVals);
    const lbl = document.createElement('div'); lbl.className='hm-label'; lbl.textContent=r;
    grid.appendChild(lbl);
    rowVals.forEach(v=>{
      const cell = document.createElement('div');
      cell.className='hm-cell';
      cell.style.background = heatColor(v, min, max, negOk);
      cell.textContent = valueFmt(v);
      cell.title = `${r}: ${valueFmt(v)}`;
      grid.appendChild(cell);
    });
  });
  el.appendChild(grid);
}
